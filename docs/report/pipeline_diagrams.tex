\section{Pipeline and architecture diagrams}\label{sec:diagrams}
\newcommand{\Wmat}{\bm{W}}  % observation weight matrix (network design rows)
% ============================================================
%  Pipeline and architecture diagrams (curated TikZ).
%
%  These are hand-authored process diagrams: they show data flow and
%  algorithm phases, which do not fall out of a parser.  Keep them in
%  sync with the code by hand; each box cites the function it maps to.
%  (The machine-generated *reference tables* live in the appendix and
%  are produced by scripts/gen_report_tables.py instead.)
%
%  Shared TikZ styles (proc/io/store/note/flow/feedback) and the
%  \fitwidth helper are defined in main.tex.
% ============================================================

This section sketches the control- and data-flow structures that the prose
chapters describe algebraically: the per-epoch estimation loop
(\Cref{fig:epoch-loop}), the shared geometry kernel behind the three
differencing regimes (\Cref{fig:differencing}), the real-time SSR correction
pipeline (\Cref{fig:ssr-pipeline}), the real-time streaming front-end that
feeds it (\Cref{fig:rt-stream}), and the PPP-RTK network producer's joint
estimator (\Cref{fig:net-flow}).  Boxes name the function or class that
realises each step, so the diagrams double as a map into \texttt{src/}.

\caption{Real-time streaming front-end ahead of the SSR correction stage of
\Cref{fig:ssr-pipeline}.  Raw bytes from \texttt{NtripClient}/\texttt{TcpClient}
are reassembled into CRC-validated RTCM3 frames by \texttt{RtcmStreamReader},
then decoded and dispatched by \texttt{feedRtcmFrame}: MSM observations
accumulate in an edge-triggered epoch buffer, while SSR corrections and
broadcast ephemerides flow into their stores.  When
\texttt{enable\_obs\_ssr\_sync} is set, an obs epoch that runs ahead of the SSR
coverage horizon (newest SSR content epoch $+$ \texttt{ssr\_max\_age}) is held
in \texttt{held\_obs\_} and re-attempted by \texttt{drainHeldObs} once an SSR
push advances the horizon; otherwise it is passed straight to
\texttt{processEpoch}.  See \dcref{sec:rt}{the real-time engine}.}

\begin{figure}[tbp]
\centering
\fitwidth{%
\begin{tikzpicture}[node distance=7mm and 10mm]
  \node[io, text width=4.6cm] (obs)
        {Multi-station observations\\{\footnotesize \texttt{MultiRinexSync} (by station)}};
  \node[proc, text width=6.2cm, below=of obs] (layout)
        {Register shared sat parameters\\{\footnotesize \texttt{registerClock} $\cdot$ \texttt{registerCodeBias} $\cdot$ \texttt{registerPhaseBias}}};
  \node[proc, text width=6.2cm, below=of layout] (begin)
        {Open epoch arrowhead system\\{\footnotesize \texttt{NetworkEstimator::beginEpoch}}};
  \node[proc, text width=6.2cm, below=of begin] (add)
        {Add per-station design rows\\{\footnotesize \texttt{addStation} ($\jacH_s$, $\jacH_r$, $\Wmat$, $\bm{y}$); one block per station}};
  \node[proc, text width=6.2cm, below=of add] (schur)
        {Schur-reduce local nuisances\\{\footnotesize \texttt{NetworkNormalEquations} arrowhead}};
  \node[proc, text width=6.2cm, below=of schur] (solve)
        {S-basis datum $+$ solve shared\\{\footnotesize \texttt{solve(Datum)} $\cdot$ \texttt{solveSharedConstrained}}};
  \node[proc, text width=6.2cm, below=of solve] (map)
        {Map to solution epoch\\{\footnotesize \texttt{toSolutionEpoch}}};
  \node[io, text width=6.2cm, below=of map] (emit)
        {SSR-style products\\{\footnotesize \texttt{NetworkProductEmitter::emitToStore} (IGS-SSR convention)}};

  \node[note, right=9mm of add, text width=3.6cm]
        {shared cols $=$ (sat, kind, signal); local cols $=$ per-station nuisances (rx clock, ZWD, iono, ambiguities)};
  \node[note, right=9mm of solve, text width=3.6cm]
        {datum pins the gauge freedoms: overall clock, the phase-bias$\leftrightarrow$ambiguity datum, ISB};

  \foreach \a/\b in {obs/layout, layout/begin, begin/add, add/schur,
                     schur/solve, solve/map, map/emit}
      \draw[flow] (\a) -- (\b);
  \draw[feedback] (emit.east) -- ++(1.4,0) |- (obs.east)
        node[pos=0.25, right, note] {next epoch};
\end{tikzpicture}}
\caption{PPP-RTK network producer joint estimator (\texttt{NetworkEstimator}).
Each epoch the shared satellite-side columns---one per
(satellite, parameter-kind, signal) for the estimable clock, code bias, and
phase bias---are registered once, then \texttt{beginEpoch} opens an arrowhead
normal-equation system of that width.  \texttt{addStation} contributes one
pre-formed block of local-vs-shared design rows per station (the network
applies no receiver-side differencing); the local nuisances are Schur-reduced
out, leaving a reduced shared system that is rank-deficient by the gauge
freedoms.  \texttt{solve} pins those with an explicit minimum-constraint
S-basis datum on a copy of the reduced system, and \texttt{toSolutionEpoch}
maps the shared estimate to an emittable record that
\texttt{NetworkProductEmitter} serializes in IGS-SSR convention.  The full
\texttt{Application::runNetwork()} driver runs this core in a live per-epoch
loop over the synced \texttt{MultiRinexSync} stations, with a two-pass station
assembly (\texttt{networkPassAHead\_}/\texttt{networkPassB\_}) feeding
\texttt{addStation}; when \texttt{network.closed\_loop} is armed it additionally
drives an undifferenced PPP-RTK consumer against the freshly emitted products.
See \dcref{sec:net}{the network mode}.}
\label{fig:net-flow}
\end{figure}
